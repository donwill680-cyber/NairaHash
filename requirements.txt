from __future__ import annotations

import asyncio
from datetime import datetime

from sqlalchemy import select

from app.db import AsyncSessionLocal
from app.models import RewardRule, User


async def process_pending_daily_rewards() -> None:
    async with AsyncSessionLocal() as session:
        rule = await session.execute(select(RewardRule).where(RewardRule.rule_type == "daily_reward", RewardRule.active.is_(True)))
        daily_rule = rule.scalar_one_or_none()
        if not daily_rule:
            return

        users = await session.execute(select(User).where(User.is_active.is_(True)))
        for user in users.scalars().all():
            # prevent duplicate processing by source key, reward credit logic is handled by the ledger service
            from app.models import UserRewardLedger

            existing = await session.execute(
                select(UserRewardLedger).where(
                    UserRewardLedger.user_id == user.id,
                    UserRewardLedger.source_type == "daily_reward",
                    UserRewardLedger.source_ref == f"scheduled-{datetime.utcnow().date().isoformat()}",
                )
            )
            if existing.scalar_one_or_none():
                continue

            user.demo_balance_sats += daily_rule.amount_sats
            user.total_rewards_sats += daily_rule.amount_sats
            ledger = UserRewardLedger(
                user_id=user.id,
                source_type="daily_reward",
                source_ref=f"scheduled-{datetime.utcnow().date().isoformat()}",
                delta_sats=daily_rule.amount_sats,
                balance_after_sats=user.demo_balance_sats,
                status="credited",
                description="Scheduled daily reward processing",
            )
            session.add(ledger)
        await session.commit()


def run_reward_cycle() -> None:
    asyncio.run(process_pending_daily_rewards())
